// Portable preparation wrapper. Executed worker originals remain byte-for-byte .txt evidence.
import fs from "node:fs";
import { resolve } from "node:path";
import { createHash } from "node:crypto";
import { execFileSync } from "node:child_process";
import { pathToFileURL } from "node:url";
const args = process.argv.slice(2);
const option = (name) => {
  const index = args.indexOf(name);
  if (index < 0 || !args[index + 1]) throw new Error(`${name} is required`);
  return resolve(args[index + 1]);
};
const repository = option("--repository");
const evidence = option("--evidence");
const output = option("--output");
const sha = (bytes) => createHash("sha256").update(bytes).digest("hex");
const binding = JSON.parse(fs.readFileSync(resolve(evidence, "source-binding.json")));
const originalBase = "/workspace/work/merge10-implementation/trie-prototype";
fs.mkdirSync(output); // Refuse to mix experiments or overwrite an existing directory.
for (const variant of ["old", "chunk2"]) {
  const root = resolve(output, variant);
  for (const file of binding.files) {
    const bytes = execFileSync("git", ["-C", repository, "show", `${binding.sourceRevision}:${file.relative}`], { maxBuffer: 32 * 1024 * 1024 });
    if (bytes.length !== file.bytes || sha(bytes) !== file.sha256) throw new Error(`historical source mismatch: ${file.relative}`);
    const target = resolve(root, file.relative);
    fs.mkdirSync(resolve(target, ".."), { recursive: true });
    fs.writeFileSync(target, bytes);
  }
  const profile = "qualification/step3/comparison/browser-reference-profile.json";
  fs.writeFileSync(resolve(root, profile), execFileSync("git", ["-C", repository, "show", `${binding.sourceRevision}:${profile}`]));
  fs.symlinkSync(resolve(repository, "node_modules"), resolve(root, "node_modules"), "dir");
}
execFileSync("git", ["apply", resolve(evidence, "prototype.patch")], { cwd: resolve(output, "chunk2") });
for (const file of ["compare-worker.mjs", "trace-worker-v2.mjs", "trace-runner-v2.mjs", "run-comparisons.py", "run-traces-v2.py"]) {
  let source = fs.readFileSync(resolve(evidence, `${file}.txt`), "utf8").replaceAll(originalBase, output);
  source = source.replaceAll("/workspace/coedit/scripts/step3-native-trace-counter.mjs", resolve(output, "step3-native-trace-counter.mjs"));
  fs.writeFileSync(resolve(output, file), source);
}
fs.copyFileSync(resolve(evidence, "step3-native-trace-counter.mjs.txt"), resolve(output, "step3-native-trace-counter.mjs"));
for (const variant of ["old", "chunk2"]) {
  const { sourceIdentity } = await import(pathToFileURL(resolve(output, variant, "scripts/step3-native-publication-worker.mjs")));
  const actual = sourceIdentity();
  const expected = JSON.parse(fs.readFileSync(resolve(evidence, `${variant}-manifest.json`)));
  for (const field of ["sources", "automergeVersion", "nativeJavaScript", "nativeWasm"]) {
    if (JSON.stringify(actual[field]) !== JSON.stringify(expected[field])) throw new Error(`${variant} ${field} does not match recorded runtime`);
  }
  fs.writeFileSync(resolve(output, `${variant}-reproduced-manifest.json`), JSON.stringify(actual, null, 2) + "\n");
}
const files = ["compare-worker.mjs", "trace-worker-v2.mjs", "trace-runner-v2.mjs", "run-comparisons.py", "run-traces-v2.py", "step3-native-trace-counter.mjs"];
fs.writeFileSync(resolve(output, "reproduction-harness.json"), JSON.stringify({ sourceRevision: binding.sourceRevision, node: process.version, argv: process.argv, files: files.map((path) => ({ path, sha256: sha(fs.readFileSync(resolve(output, path))) })) }, null, 2) + "\n");
console.log(`Prepared and verified both isolated variants in ${output}`);
if (args.includes("--run-comparisons") || args.includes("--run-traces")) execFileSync("python3", [resolve(output, "run-comparisons.py")], { stdio: "inherit" });
if (args.includes("--run-traces")) execFileSync("python3", [resolve(output, "run-traces-v2.py")], { stdio: "inherit" });
