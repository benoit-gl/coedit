import pathlib,subprocess,json
base=pathlib.Path('/workspace/work/merge10-packed-prototype')
with (base/'measurements-h500.jsonl').open('x') as journal:
 for variant in ['trie','unit1','commandPack']:
  command=['node','--expose-gc',str(base/'measure500-worker.mjs'),variant,str(base/'history500.json'),str(base/f'{variant}-h500.am')]
  journal.write(json.dumps({'phase':'start','variant':variant,'command':command})+'\n');journal.flush()
  with (base/f'{variant}-h500.stdout.log').open('x') as stdout,(base/f'{variant}-h500.stderr.log').open('x') as stderr:
   result=subprocess.run(command,text=True,stdout=stdout,stderr=stderr)
  lines=(base/f'{variant}-h500.stdout.log').read_text().splitlines()
  for line in lines:
   row=json.loads(line);row['command']=command;journal.write(json.dumps(row)+'\n');journal.flush()
  if result.returncode:
   journal.write(json.dumps({'phase':'failed','variant':variant,'exitCode':result.returncode})+'\n');journal.flush()
   print(json.dumps({'variant':variant,'failed':result.returncode}),flush=True)
  else:print(json.dumps({k:row.get(k) for k in ['variant','complete','encodedBytes','nativeOps','lastCommandOps','heapUsedDelta','historicalRootsChecked']}),flush=True)
